import json
import os
from pathlib import Path
from urllib import request, error
from storage.history import SolveHistory
from storage.sync import build_summary_payload

url = os.environ.get("RUBIKS_SYNC_URL")
key = os.environ.get("RUBIKS_SYNC_API_KEY")
pending = SolveHistory(Path("data/solve_history.db")).pending()

print("URL:", url or "NOT SET")
print("API key:", "SET" if key else "NOT SET")
print("Pending:", len(pending))

if not url or not key or not pending:
    raise SystemExit()

req = request.Request(
    url,
    data=json.dumps(build_summary_payload(pending[0])).encode(),
    method="POST",
    headers={
        "Content-Type": "application/json",
        "X-API-Key": key,
    },
)

try:
    with request.urlopen(req, timeout=10) as response:
        print("HTTP:", response.status)
        print("Response:", response.read().decode())
except error.HTTPError as exc:
    print("HTTP:", exc.code)
    print("Response:", exc.read().decode())
except error.URLError as exc:
    print("Connection error:", exc.reason)
PY
